Below is a deployment plan for installing Prometheus + Node Exporter + Grafana on the Ubuntu 24.04 OpenCenter server, exposing the OpenCenter /metrics exporter, and provisioning the OpenCenter dashboards.

/home/dzoan/OSPC2FLEX/osflex-deployer-fullmig-5.0.0420current

#!/usr/bin/env bash
set -Eeuo pipefail

PROJECT_ROOT="${PROJECT_ROOT:-/home/dzoan/OSPC2FLEX/osflex-deployer-fullmig-5.0.0420current}"

SOURCE_ROOT="$PROJECT_ROOT/infrastructure/monitoring/grafana"

  "opencenter-cluster-overview.json"
  "opencenter-deployment.json"
  "opencenter-gitops.json"
  "opencenter-infrastructure.json"

  -m 0750 \
  "$DASHBOARD_DEST"

for dashboard in "${required_dashboards[@]}"; do

    -g grafana \
    -m 0640 \

    -o root \
    -g grafana \
    -m 0640 \

7. Put Prometheus and Grafana behind Nginx

Do not expose Prometheus directly to the Internet.

Install Nginx:

Create an OpenCenter reverse proxy:

sudo tee /etc/nginx/sites-available/opencenter-monitoring >/dev/null <<'EOF'
map $http_upgrade $connection_upgrade {
    default upgrade;
    '' close;
}

server {
    listen 80;
    server_name _;

    client_max_body_size 25m;

    location /grafana/ {
        proxy_pass http://127.0.0.1:3000/;
        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection $connection_upgrade;
    }

    location /prometheus/ {
        auth_basic "OpenCenter Prometheus";
        auth_basic_user_file /etc/nginx/.prometheus-htpasswd;

        proxy_pass http://127.0.0.1:9090/;
        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location /opencenter-exporter-health {
        proxy_pass http://127.0.0.1:9187/healthz;
        proxy_set_header Host $host;
    }

    # Replace this port with the existing OpenCenter dashboard port.
    location / {
        proxy_pass http://127.0.0.1:8080;
        proxy_http_version 1.1;

        proxy_buffering off;
        proxy_cache off;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection $connection_upgrade;

}
