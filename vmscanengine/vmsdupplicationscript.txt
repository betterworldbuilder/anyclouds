#!/usr/bin/env bash
# R6 Containerization Discovery Script
# Read-only capture from live VM evidence for Docker/Kubernetes generation.
set -uo pipefail

# ================================================================
# OSPC/FLEX VM → R6 Containerization Evidence Capture
# ────────────────────────────────────────────────────────────────
# Goal         : capture all application/runtime/config/process/data
#                evidence needed to build working containers in K8s
# Mode         : READ-ONLY CAPTURE (no FLEX mutation, no VM clone)
# Output       : ./migration-csv/UnknownCustomer-audit/<node>/r6-container-discovery/
# Security     : redact likely secrets; record private key paths only
# ================================================================

DRY_RUN=${DRY_RUN:-0}
DEFAULT_SRC_USER="ubuntu"
SSH_KEY="/home/dzoan/.ssh/id_rsa"
SSH_OPTS="-i $SSH_KEY -o StrictHostKeyChecking=no -o ConnectTimeout=15 -o BatchMode=yes"
AUDIT_DIR="./migration-csv/UnknownCustomer-audit"
APP_PATHS="${APP_PATHS:-/opt/app /srv /var/www /var/lib/app /home}"
mkdir -p "$AUDIT_DIR"

trap 'pkill -P $$ 2>/dev/null || true' EXIT HUP INT TERM

run() {
  if [ "$DRY_RUN" = "1" ]; then
    echo "  [DRY] $*"
  else
    "$@"
  fi
}

src_cmd() { ssh $SSH_OPTS "$SRC_USER@$SRC_IP" "$@" 2>/dev/null; }

copy_from_src() {
  local remote_path="$1"
  local local_path="$2"
  scp -q $SSH_OPTS "$SRC_USER@$SRC_IP:$remote_path" "$local_path" 2>/dev/null
}

capture_r6_container_evidence() {
  local NODE_NAME="$1"
  local NODE_AUDIT="$2"
  local R6_DIR="$NODE_AUDIT/r6-container-discovery"
  local REMOTE_TGZ="~/r6-container-discovery-${NODE_NAME}.tgz"
  mkdir -p "$R6_DIR"

  echo ""
  echo "══════════════════════ R6 -- CONTAINERIZATION DISCOVERY ══════════════════════"
  echo "[R6] Node: $NODE_NAME"
  echo "[R6] Source VM: $SRC_IP [$SRC_USER]"
  echo "[R6] Output: $R6_DIR"
  echo "[R6] Capturing read-only evidence for Dockerfiles, K8s manifests, ConfigMaps, Secrets, PVCs, Services, probes, CronJobs/workers, and dependency contracts."

  ssh $SSH_OPTS "$SRC_USER@$SRC_IP" "APP_PATHS='$APP_PATHS' NODE_NAME='$NODE_NAME' bash -s" <<'R6REMOTE'
set +e
NODE_NAME="${NODE_NAME:-unknown-node}"
APP_PATHS="${APP_PATHS:-/opt/app /srv /var/www /var/lib/app /home}"
DISC="$HOME/r6-container-discovery-${NODE_NAME}"
mkdir -p "$DISC"/00_summary "$DISC"/01_system "$DISC"/02_processes "$DISC"/03_services "$DISC"/04_ports_network "$DISC"/05_apps "$DISC"/06_dependencies/manifests "$DISC"/07_config "$DISC"/08_data_volumes "$DISC"/09_external_deps "$DISC"/10_tls "$DISC"/11_k8s "$DISC"/12_security "$DISC"/13_container_contract

redact_stream() {
  sed -E 's/((PASSWORD|PASS|SECRET|TOKEN|API[_-]?KEY|PRIVATE[_-]?KEY|CREDENTIAL|DATABASE_URL|REDIS_URL|AWS_SECRET|AZURE_CLIENT_SECRET|GCP_KEY)[A-Za-z0-9_ .:-]*[=:][[:space:]]*)[^[:space:]#"'\''']+/\1<REDACTED>/Ig'
}

redact_file() {
  [ -f "$1" ] && redact_stream < "$1" 2>/dev/null
}

json_escape() {
  sed 's/\\/\\\\/g; s/"/\\"/g' | tr -d '\r'
}

OS_PRETTY="$(. /etc/os-release 2>/dev/null; echo "${PRETTY_NAME:-unknown}")"
HOSTNAME_NOW="$(hostname 2>/dev/null || echo unknown)"
{
  echo "{"
  echo "  \"schema\": \"r6-containerization-evidence/v1\","
  echo "  \"captured_at\": \"$(date -Is)\","
  echo "  \"node\": \"$(printf '%s' "$NODE_NAME" | json_escape)\","
  echo "  \"hostname\": \"$(printf '%s' "$HOSTNAME_NOW" | json_escape)\","
  echo "  \"os\": \"$(printf '%s' "$OS_PRETTY" | json_escape)\","
  echo "  \"kernel\": \"$(uname -r 2>/dev/null | json_escape)\","
  echo "  \"purpose\": \"Generate Dockerfiles and Kubernetes manifests from live VM application evidence\","
  echo "  \"must_not_bake_into_images\": [\"ssh keys\", \"password hashes\", \"TLS private keys\", \"host network config\", \"kernel settings\", \"cloud agent identity\", \"raw secrets\"]"
  echo "}"
} > "$DISC/00_summary/containerization-source-summary.json"

# 01 system facts: useful for base-image/runtime inference, not VM cloning.
hostnamectl > "$DISC/01_system/hostnamectl.txt" 2>/dev/null || true
cat /etc/os-release > "$DISC/01_system/os-release.txt" 2>/dev/null || true
uname -a > "$DISC/01_system/uname.txt" 2>/dev/null || true
lscpu > "$DISC/01_system/lscpu.txt" 2>/dev/null || true
free -m > "$DISC/01_system/memory-mb.txt" 2>/dev/null || true
df -Th > "$DISC/01_system/filesystems.txt" 2>/dev/null || true
lsblk -f > "$DISC/01_system/block-devices.txt" 2>/dev/null || true

# 02 processes: exact commands, working dirs, and process ownership for ENTRYPOINT/CMD.
ps auxww > "$DISC/02_processes/processes-auxww.txt" 2>/dev/null || true
pstree -ap > "$DISC/02_processes/process-tree.txt" 2>/dev/null || true
: > "$DISC/02_processes/process-cwd-cmd-user.tsv"
for proc in /proc/[0-9]*; do
  pid="${proc##*/}"
  [ -r "$proc/cmdline" ] || continue
  cmd="$(tr '\0' ' ' < "$proc/cmdline" 2>/dev/null | sed 's/[[:space:]]*$//')"
  [ -n "$cmd" ] || continue
  cwd="$(readlink "$proc/cwd" 2>/dev/null || true)"
  user="$(stat -c '%U' "$proc" 2>/dev/null || true)"
  printf '%s\t%s\t%s\t%s\n' "$pid" "$user" "$cwd" "$cmd" >> "$DISC/02_processes/process-cwd-cmd-user.tsv"
done

# 03 services: systemd-to-container hints.
systemctl list-units --type=service --state=running --no-pager > "$DISC/03_services/running-services.txt" 2>/dev/null || true
systemctl list-unit-files --type=service --state=enabled --no-pager > "$DISC/03_services/enabled-services.txt" 2>/dev/null || true
systemctl list-unit-files --type=service --state=enabled --no-legend 2>/dev/null | awk '{print $1}' \
  | grep -Ev '^(ssh|sshd|systemd-|dbus|snapd|cloud-init|ufw|cron|rsyslog|network|NetworkManager|multipathd|open-vm|nova-agent|rackspace|holland)' \
  > "$DISC/03_services/container-candidate-services.txt" || true
while read svc; do
  [ -n "$svc" ] || continue
  safe="$(echo "$svc" | tr '/:@ ' '____')"
  systemctl cat "$svc" > "$DISC/03_services/${safe}.unit.txt" 2>/dev/null || true
  systemctl show "$svc" \
    -p Id -p Description -p LoadState -p ActiveState -p SubState -p Type -p MainPID \
    -p User -p Group -p WorkingDirectory -p RootDirectory \
    -p ExecStart -p ExecStartPre -p ExecStartPost -p ExecReload \
    -p Environment -p EnvironmentFiles -p Restart -p RestartSec \
    > "$DISC/03_services/${safe}.container-hints.txt" 2>/dev/null || true
  pid="$(systemctl show "$svc" -p MainPID --value 2>/dev/null)"
  if [ -n "$pid" ] && [ "$pid" != "0" ] && [ -r "/proc/$pid/environ" ]; then
    tr '\0' '\n' < "/proc/$pid/environ" 2>/dev/null | redact_stream > "$DISC/03_services/${safe}.redacted-runtime-env.txt"
  fi
done < "$DISC/03_services/container-candidate-services.txt"

# 04 networking: service ports and routing context for Service/Ingress generation.
ss -lntup > "$DISC/04_ports_network/listening-ports-processes.txt" 2>/dev/null || true
ss -ntup > "$DISC/04_ports_network/active-connections.txt" 2>/dev/null || true
ip addr > "$DISC/04_ports_network/ip-addr.txt" 2>/dev/null || true
ip route > "$DISC/04_ports_network/ip-route.txt" 2>/dev/null || true
cat /etc/hosts > "$DISC/04_ports_network/etc-hosts.txt" 2>/dev/null || true
getent hosts > "$DISC/04_ports_network/hosts-resolution.txt" 2>/dev/null || true

# 05/06 app roots and dependency manifests: base images and build commands.
: > "$DISC/05_apps/app-root-candidates.txt"
: > "$DISC/06_dependencies/dependency-manifest-index.txt"
for root in $APP_PATHS; do
  [ -d "$root" ] || continue
  find "$root" -maxdepth 6 -type d \( -name node_modules -o -name .git -o -name vendor -o -name __pycache__ -o -name .venv \) -prune -o -type f \( \
    -name package.json -o -name package-lock.json -o -name yarn.lock -o -name pnpm-lock.yaml -o -name .nvmrc -o -name .node-version \
    -o -name requirements.txt -o -name constraints.txt -o -name pyproject.toml -o -name poetry.lock -o -name Pipfile -o -name Pipfile.lock -o -name setup.py -o -name setup.cfg \
    -o -name composer.json -o -name composer.lock -o -name Gemfile -o -name Gemfile.lock \
    -o -name go.mod -o -name go.sum -o -name pom.xml -o -name build.gradle -o -name gradle.properties \
    -o -name Cargo.toml -o -name Cargo.lock -o -name Dockerfile -o -name docker-compose.yml -o -name docker-compose.yaml -o -name compose.yml -o -name Procfile -o -name Makefile -o -name runtime.txt \
  \) -print 2>/dev/null | while read mf; do
    echo "$(dirname "$mf")" >> "$DISC/05_apps/app-root-candidates.txt"
    echo "$mf" >> "$DISC/06_dependencies/dependency-manifest-index.txt"
  done
done
sort -u "$DISC/05_apps/app-root-candidates.txt" -o "$DISC/05_apps/app-root-candidates.txt" 2>/dev/null || true
sort -u "$DISC/06_dependencies/dependency-manifest-index.txt" -o "$DISC/06_dependencies/dependency-manifest-index.txt" 2>/dev/null || true
while read mf; do
  [ -f "$mf" ] || continue
  safe="$(echo "$mf" | sed 's#^/##; s#[/: ]#__#g')"
  cp "$mf" "$DISC/06_dependencies/manifests/$safe" 2>/dev/null || true
done < "$DISC/06_dependencies/dependency-manifest-index.txt"

# Runtime version inventory.
{
  echo "python=$(python3 --version 2>/dev/null || true)"
  echo "node=$(node --version 2>/dev/null || true)"
  echo "npm=$(npm --version 2>/dev/null || true)"
  echo "yarn=$(yarn --version 2>/dev/null || true)"
  echo "pnpm=$(pnpm --version 2>/dev/null || true)"
  echo "php=$(php --version 2>/dev/null | head -1 || true)"
  echo "composer=$(composer --version 2>/dev/null || true)"
  echo "ruby=$(ruby --version 2>/dev/null || true)"
  echo "bundle=$(bundle --version 2>/dev/null || true)"
  echo "go=$(go version 2>/dev/null || true)"
  echo "java=$(java -version 2>&1 | head -1 || true)"
  echo "docker=$(docker version --format '{{.Server.Version}}' 2>/dev/null || true)"
  echo "pm2=$(pm2 --version 2>/dev/null || true)"
} > "$DISC/06_dependencies/runtime-versions.txt"

# 07 config and secrets: values redacted, keys preserved for ConfigMap/Secret contracts.
: > "$DISC/07_config/config-file-index.txt"
: > "$DISC/07_config/redacted-config-samples.txt"
for root in $APP_PATHS /etc/nginx /etc/apache2 /etc/haproxy /etc/systemd/system; do
  [ -d "$root" ] || continue
  find "$root" -maxdepth 6 -type f \( \
    -name '.env' -o -name '*.env' -o -name '*.ini' -o -name '*.conf' -o -name '*.cfg' \
    -o -name '*.yaml' -o -name '*.yml' -o -name '*.json' -o -name '*.toml' -o -name '*.properties' \
  \) -not -path '*/node_modules/*' -not -path '*/.git/*' -print 2>/dev/null >> "$DISC/07_config/config-file-index.txt"
done
sort -u "$DISC/07_config/config-file-index.txt" -o "$DISC/07_config/config-file-index.txt" 2>/dev/null || true
head -120 "$DISC/07_config/config-file-index.txt" | while read cf; do
  [ -f "$cf" ] || continue
  echo "===== $cf =====" >> "$DISC/07_config/redacted-config-samples.txt"
  redact_file "$cf" | head -120 >> "$DISC/07_config/redacted-config-samples.txt"
done
grep -RIEh '^[[:space:]]*[A-Za-z_][A-Za-z0-9_]*(=|:)' $APP_PATHS 2>/dev/null \
  | sed -E 's/^([[:space:]]*[A-Za-z_][A-Za-z0-9_]*(=|:)).*/\1 <VALUE>/' \
  | sort -u > "$DISC/07_config/config-key-inventory.txt" || true
grep -RIEh '(PASSWORD|PASS|SECRET|TOKEN|API_KEY|PRIVATE_KEY|DATABASE_URL|REDIS_URL|AWS_|AZURE_|GCP_|MYSQL_|PGPASSWORD)' $APP_PATHS /etc/systemd/system 2>/dev/null \
  | sed -E 's/(=|:).*/\1 <REDACTED>/' \
  | sort -u > "$DISC/07_config/secret-key-candidates.txt" || true

# 08 data/writable paths: PVC or emptyDir decisions.
: > "$DISC/08_data_volumes/writable-path-candidates.txt"
for root in $APP_PATHS /var/log /var/cache /tmp; do
  [ -d "$root" ] || continue
  find "$root" -maxdepth 5 -type d \( \
    -iname '*upload*' -o -iname '*media*' -o -iname '*storage*' -o -iname '*data*' \
    -o -iname '*cache*' -o -iname '*tmp*' -o -iname '*log*' -o -perm -0002 \
  \) -print 2>/dev/null >> "$DISC/08_data_volumes/writable-path-candidates.txt"
done
sort -u "$DISC/08_data_volumes/writable-path-candidates.txt" -o "$DISC/08_data_volumes/writable-path-candidates.txt" 2>/dev/null || true
du -sh $APP_PATHS /srv/data 2>/dev/null > "$DISC/08_data_volumes/app-data-sizes.txt" || true

# 09 external dependencies/stateful services: externalize DB/cache/queues/API endpoints.
systemctl list-unit-files 2>/dev/null | grep -Ei 'postgres|mysql|mariadb|redis|mongo|rabbitmq|memcached|elastic|solr|kafka' > "$DISC/09_external_deps/local-stateful-services.txt" || true
ss -lntup 2>/dev/null | grep -E ':(5432|3306|6379|27017|5672|11211|9200|9300|8983|9092)\b' > "$DISC/09_external_deps/stateful-listening-ports.txt" || true
grep -RIE '(DB_HOST|DATABASE_URL|PGHOST|PG_HOST|MYSQL_HOST|REDIS_URL|REDIS_HOST|MONGO|RABBIT|AMQP|ELASTIC|KAFKA|API_URL|BASE_URL)' $APP_PATHS /etc/systemd/system 2>/dev/null \
  | redact_stream > "$DISC/09_external_deps/external-dependency-candidates.txt" || true
if command -v psql >/dev/null 2>&1; then
  sudo -u postgres psql -tAc "SELECT datname FROM pg_database WHERE datistemplate=false;" > "$DISC/09_external_deps/postgresql-database-names.txt" 2>/dev/null || true
fi
if command -v mysql >/dev/null 2>&1; then
  mysql -uroot -e 'SHOW DATABASES;' > "$DISC/09_external_deps/mysql-database-names.txt" 2>/dev/null || true
fi

# 10 TLS: cert metadata only. Private key paths are recorded as exclusions.
find /etc/letsencrypt /etc/ssl /etc/pki $APP_PATHS -type f \( -name '*.crt' -o -name '*.pem' \) 2>/dev/null \
  | grep -vi 'private\|privkey\|\.key$' > "$DISC/10_tls/certificate-files.txt" || true
: > "$DISC/10_tls/certificate-metadata.txt"
while read cert; do
  [ -f "$cert" ] || continue
  echo "===== $cert =====" >> "$DISC/10_tls/certificate-metadata.txt"
  openssl x509 -in "$cert" -noout -subject -issuer -dates -ext subjectAltName 2>/dev/null >> "$DISC/10_tls/certificate-metadata.txt"
done < "$DISC/10_tls/certificate-files.txt"
find /etc/letsencrypt /etc/ssl /etc/pki $APP_PATHS -type f \( -name '*.key' -o -name 'privkey*.pem' \) 2>/dev/null \
  > "$DISC/10_tls/private-key-paths-do-not-copy.txt" || true

# 11 K8s workload hints: probes, cron, workers, ingress/web-server configs.
sudo crontab -l > "$DISC/11_k8s/root-crontab.txt" 2>/dev/null || true
crontab -l > "$DISC/11_k8s/user-crontab.txt" 2>/dev/null || true
find /etc/cron.d /etc/cron.daily /etc/cron.hourly /etc/cron.monthly /etc/cron.weekly -maxdepth 2 -type f -print -exec sed -n '1,120p' {} \; \
  > "$DISC/11_k8s/system-cron-files.txt" 2>/dev/null || true
grep -RIE '(health|healthz|ready|readyz|live|livez|status|metrics|actuator)' $APP_PATHS /etc/nginx /etc/apache2 2>/dev/null \
  | head -300 > "$DISC/11_k8s/health-endpoint-hints.txt" || true
find /etc/nginx /etc/apache2 /etc/haproxy -type f -maxdepth 4 2>/dev/null > "$DISC/11_k8s/proxy-config-files.txt" || true

# 12 security: non-root UID/GID and ownership choices.
getent passwd | awk -F: '$3>=1000 && $3<65000{print}' > "$DISC/12_security/non-system-users.txt" || true
getent group | awk -F: '$3>=1000 && $3<65000{print}' > "$DISC/12_security/non-system-groups.txt" || true
find $APP_PATHS -maxdepth 3 -printf '%u:%g %m %p\n' 2>/dev/null | head -2000 > "$DISC/12_security/app-path-ownership-sample.txt" || true


# 13A production gap-fill: best-effort data that turns discovery into a deployable contract.
# These are generated as reviewable templates because final namespace, registry, secrets,
# DNS, resource policy, and storage classes are target-cluster decisions.
mkdir -p "$DISC/14_production_templates" "$DISC/15_validation" "$DISC/05_apps/build-contexts"
R6_CAPTURE_CODE="${R6_CAPTURE_CODE:-0}"
R6_TARGET_NAMESPACE="${R6_TARGET_NAMESPACE:-CHANGE_ME_NAMESPACE}"
R6_IMAGE_REGISTRY="${R6_IMAGE_REGISTRY:-CHANGE_ME_REGISTRY}"
R6_STORAGE_CLASS="${R6_STORAGE_CLASS:-CHANGE_ME_STORAGE_CLASS}"
R6_DEFAULT_REPLICAS="${R6_DEFAULT_REPLICAS:-2}"

ps -eo pid,ppid,user,group,%cpu,%mem,rss,vsz,etime,args --sort=-rss > "$DISC/02_processes/process-resource-snapshot.txt" 2>/dev/null || true

: > "$DISC/03_services/startup-shutdown-dependency-hints.txt"
while read svc; do
  [ -n "$svc" ] || continue
  safe="$(echo "$svc" | tr '/:@ ' '____')"
  systemctl show "$svc" \
    -p Id -p After -p Before -p Wants -p Requires -p PartOf -p BindsTo \
    -p ExecStop -p KillSignal -p TimeoutStartUSec -p TimeoutStopUSec \
    > "$DISC/03_services/${safe}.startup-shutdown-hints.txt" 2>/dev/null || true
  cat "$DISC/03_services/${safe}.startup-shutdown-hints.txt" >> "$DISC/03_services/startup-shutdown-dependency-hints.txt" 2>/dev/null || true
done < "$DISC/03_services/container-candidate-services.txt"

{
  echo "app_root\tstack_hint\trole_hint\treason"
  while read app; do
    [ -d "$app" ] || continue
    stack="unknown"; role="unknown"; reason=""
    [ -f "$app/package.json" ] && stack="nodejs" && reason="$reason package.json"
    { [ -f "$app/requirements.txt" ] || [ -f "$app/pyproject.toml" ] || [ -f "$app/Pipfile" ]; } && stack="python" && reason="$reason python-manifest"
    [ -f "$app/composer.json" ] && stack="php" && reason="$reason composer.json"
    [ -f "$app/Gemfile" ] && stack="ruby" && reason="$reason Gemfile"
    [ -f "$app/go.mod" ] && stack="go" && reason="$reason go.mod"
    { [ -f "$app/pom.xml" ] || [ -f "$app/build.gradle" ]; } && stack="java" && reason="$reason java-manifest"
    grep -RIEq "(listen|server_name|routes|express|fastapi|flask|django|rails|spring|gunicorn|uvicorn|uwsgi|nginx|apache)" "$app" 2>/dev/null && role="web"
    grep -RIEq "(worker|queue|celery|rq|sidekiq|bull|consumer|supervisor)" "$app" 2>/dev/null && role="$role,worker"
    grep -RIEq "(cron|schedule|scheduler|beat)" "$app" 2>/dev/null && role="$role,scheduler"
    printf "%s\t%s\t%s\t%s\n" "$app" "$stack" "$role" "$reason"
  done < "$DISC/05_apps/app-root-candidates.txt"
} > "$DISC/05_apps/app-classification.tsv"

find $APP_PATHS -maxdepth 6 -type f \
  -not -path '*/node_modules/*' -not -path '*/.git/*' -not -path '*/vendor/*' -not -path '*/__pycache__/*' \
  -printf '%s\t%TY-%Tm-%TdT%TH:%TM:%TS\t%p\n' 2>/dev/null \
  | sort -nr | head -5000 > "$DISC/05_apps/build-context-file-inventory.tsv" || true

if [ "$R6_CAPTURE_CODE" = "1" ]; then
  while read app; do
    [ -d "$app" ] || continue
    safe="$(echo "$app" | sed 's#^/##; s#[/: ]#__#g')"
    tar --exclude='.git' --exclude='node_modules' --exclude='vendor' --exclude='__pycache__' \
      --exclude='*.key' --exclude='id_rsa*' --exclude='*.pem' --exclude='.env' \
      -czf "$DISC/05_apps/build-contexts/${safe}.sanitized-source.tgz" -C "$(dirname "$app")" "$(basename "$app")" 2>/dev/null || true
  done < "$DISC/05_apps/app-root-candidates.txt"
fi

dpkg-query -W -f='${binary:Package}\t${Version}\n' > "$DISC/06_dependencies/os-package-inventory.tsv" 2>/dev/null || true
apt-mark showmanual > "$DISC/06_dependencies/apt-manual-packages.txt" 2>/dev/null || true
ldconfig -p > "$DISC/06_dependencies/native-library-cache.txt" 2>/dev/null || true

awk -F'[=: ]+' '
  BEGIN { print "# Fill values from vault/secret manager; do not commit real values." }
  /^[[:space:]]*$/ { next }
  {
    for (i=1; i<=NF; i++) {
      if ($i ~ /(PASSWORD|PASS|SECRET|TOKEN|API_KEY|PRIVATE_KEY|DATABASE_URL|REDIS_URL|AWS_|AZURE_|GCP_|MYSQL_|PGPASSWORD)/) {
        gsub(/[^A-Za-z0-9_]/, "", $i)
        if ($i != "") print $i"="
      }
    }
  }' "$DISC/07_config/secret-key-candidates.txt" | sort -u > "$DISC/07_config/secrets.required.env.template" || true
awk -F'[=: ]+' '
  BEGIN { print "# Non-secret config keys. Review and move sensitive keys to secrets.required.env.template." }
  /^[[:space:]]*$/ { next }
  {
    key=$1; gsub(/^[[:space:]]+|[[:space:]]+$/, "", key); gsub(/[^A-Za-z0-9_]/, "", key)
    if (key != "" && key !~ /(PASSWORD|PASS|SECRET|TOKEN|API_KEY|PRIVATE_KEY|DATABASE_URL|REDIS_URL)/) print key"="
  }' "$DISC/07_config/config-key-inventory.txt" | sort -u > "$DISC/07_config/configmap.env.template" || true

{
  echo "path\trecommendation\treason"
  while read path; do
    [ -n "$path" ] || continue
    rec="emptyDir"; reason="temporary/cache/log-like path"
    echo "$path" | grep -Eiq 'upload|media|storage|data' && rec="PersistentVolumeClaim" && reason="user/application data path"
    echo "$path" | grep -Eiq '/var/log|/cache|/tmp' && rec="emptyDir-or-sidecar-log" && reason="ephemeral/cache/log path"
    printf "%s\t%s\t%s\n" "$path" "$rec" "$reason"
  done < "$DISC/08_data_volumes/writable-path-candidates.txt"
} > "$DISC/08_data_volumes/volume-recommendations.tsv"

grep -RIEho '([a-zA-Z0-9_.-]+\.)+[a-zA-Z]{2,}(:[0-9]+)?' $APP_PATHS /etc/nginx /etc/apache2 /etc/haproxy 2>/dev/null \
  | sort -u > "$DISC/09_external_deps/domain-endpoint-candidates.txt" || true

{
  echo "# server_name / ServerName / ServerAlias candidates"
  grep -RIEh '^[[:space:]]*(server_name|ServerName|ServerAlias)[[:space:]]+' /etc/nginx /etc/apache2 2>/dev/null | sed -E 's/[;]//g' | sort -u

{
  echo "port\tpath\thttp_code"
  ports="$(ss -lnt 2>/dev/null | awk 'NR>1{print $4}' | sed -E 's/.*:([0-9]+)$/\1/' | grep -E '^(80|443|3000|5000|5001|8000|8080|8081|9000)$' | sort -u)"
  for port in $ports; do
    scheme="http"; [ "$port" = "443" ] && scheme="https"
    for path in /health /healthz /ready /readyz /live /livez /status /metrics /; do
      code="$(curl -k -sS -o /dev/null -w '%{http_code}' --max-time 3 "$scheme://127.0.0.1:$port$path" 2>/dev/null || echo 000)"
      printf "%s\t%s\t%s\n" "$port" "$path" "$code"
    done
  done
} > "$DISC/11_k8s/local-probe-tests.tsv" || true
