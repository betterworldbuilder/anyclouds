Component type detection:

function getComponentDeviceType(component) {
  const role = `${component.role} ${component.component_type}`.toLowerCase();

  if (role.includes("frontend") || role.includes("mobile") || role.includes("web")) {
    return "mobile_phone";
  }

  if (role.includes("api gateway") || role.includes("gateway")) {
    return "api_console_tablet";
  }

  if (role.includes("api server") || role.includes("service")) {
    return "service_console";
  }

  if (role.includes("backend") || role.includes("core") || role.includes("ledger")) {
    return "backend_engine_rack";
  }

  if (role.includes("auth") || role.includes("sso") || role.includes("identity")) {
    return "auth_security_device";
  }

  if (role.includes("database") || role.includes("postgres") || role.includes("mysql") || role.includes("db")) {
    return "database_console";
  }

  if (role.includes("cache") || role.includes("redis")) {
    return "cache_memory_chip";
  }

  if (role.includes("queue") || role.includes("rabbit") || role.includes("kafka")) {
    return "queue_conveyor";
  }

  if (role.includes("worker") || role.includes("cron") || role.includes("batch")) {
    return "worker_factory";
  }

  if (role.includes("monitor") || role.includes("observability") || role.includes("logging")) {
    return "monitoring_noc_wall";
  }

  return "service_console";
}

Generic HTTP source-vs-target test:

function buildHttpCompareCommand(component, path) {
  return `
SOURCE_URL="${component.source_url}"
TARGET_URL="${component.target_url}"
PATH_TO_TEST="${path}"
TIMEOUT=5

echo "Testing SOURCE: ${SOURCE_URL}${PATH_TO_TEST}"
curl -k -sS -o /tmp/source.out -w "source_http=%{http_code} source_time=%{time_total} source_ip=%{remote_ip}\\n" --max-time "$TIMEOUT" "${SOURCE_URL}${PATH_TO_TEST}"

echo "Testing TARGET: ${TARGET_URL}${PATH_TO_TEST}"
curl -k -sS -o /tmp/target.out -w "target_http=%{http_code} target_time=%{time_total} target_ip=%{remote_ip}\\n" --max-time "$TIMEOUT" "${TARGET_URL}${PATH_TO_TEST}"

echo "--- SOURCE BODY PREVIEW ---"
head -20 /tmp/source.out || true

echo "--- TARGET BODY PREVIEW ---"
head -20 /tmp/target.out || true
`;
}

Result parsing:
Parse curl output:
- http_code
- time_total
- remote_ip

PASS rules:
- HTTP health: source and target return 200, 204, 301, or 302
- Ready: target returns 200
- DB SELECT 1 succeeds
- Redis PING returns PONG
- Queue management health returns OK
- Worker process exists if worker is in scope
- Monitoring command returns CPU/RAM/disk output

WARNING rules:
- target works but source unavailable and operator approved
- target latency is more than 25% slower than source
- optional endpoint missing
- cache session strategy not selected
- queue depth above warning threshold but below blocker threshold

BLOCKED rules:
- critical component target unavailable
- target health fails
- DB SELECT 1 fails
- queue unavailable when in scope
- duplicate workers active on source and target during final cutover
- source hostname cannot resolve when source comparison is required

Integrate with Stage UAT gates:

- update the component row status
- update Component Health Gate
- update DB Gate if DB component
- update Cache Gate if cache component
- update Queue Gate if queue component
- update Worker Gate if worker component

Gate behavior:
- If any critical component test is BLOCKED, block Start Cutover.
- If DB component is in scope and DB tests fail, block Ramp to 100%.
- If Cache in scope and cache tests fail, block Ramp to 100%.
- If Queue in scope and queue tests fail, block Ramp to 100%.
- If business transaction validation fails, block Complete Cutover.
